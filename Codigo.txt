PLANTEAMIENTO DEL EJERCICIO
14. TIENDA DE SNEAKERS Y ROPA URBANA

En el módulo de inventario y ventas de una tienda streetwear (clase mayor: mStore) se procesan los artículos vendidos (clase menor base: mPrenda), conociendo el código, la marca y el costo de adquisición, sobre el cual se calcula un precio inicial aplicando un margen base del 30%. Se comercializan dos líneas especializadas:
- Sneakers Edición Limitada (mSneaker): al precio inicial le suma $15 de tasa de importación; si es una talla de alta demanda, incrementa un 10% adicional sobre ese subtotal.
- Hoodie / Sudadera (mHoodie): al precio inicial le suma $8 si incluye estampado personalizado; si pertenece a una colección pasada, aplica un descuento de $6 sobre el monto resultante.

Se desea determinar el precio final de venta y la ganancia neta (precioFinal - costo) de cada producto, la ganancia total obtenida por la tienda y el código del producto que generó la mayor ganancia neta.

Nuevo Requerimiento:
- Atributo en clase menor derivada (mSneaker) -> -> Cálculo de menor en clase mayor (mStore): Incluir el atributo tallaUs (real) en mSneaker para indicar la talla numérica vendida, e implementar un método menorTallaSneaker() en mStore que retorne la menor talla registrada entre los sneakers vendidos.

Datos de entrada:
- Código: S01, Marca: Nike, Costo: 50, Tipo: Sneaker, Alta demanda: Sí, Talla US: 9.5
- Código: H01, Marca: Adidas, Costo: 20, Tipo: Hoodie, Estampado: Sí, Colección pasada: No
- Código: S02, Marca: Puma, Costo: 40, Tipo: Sneaker, Alta demanda: No, Talla US: 8.5
- Código: H02, Marca: Vans, Costo: 30, Tipo: Hoodie, Estampado: No, Colección pasada: Sí

La salida requerida presenta el siguiente formato:
S01 (Nike) -> Precio: $88.00 | Ganancia: $38.00
H01 (Adidas) -> Precio: $34.00 | Ganancia: $14.00
S02 (Puma) -> Precio: $67.00 | Ganancia: $27.00
H02 (Vans) -> Precio: $33.00 | Ganancia: $3.00

Ganancia total de la tienda: $82.00
Producto con mayor ganancia neta: S01
Menor talla de sneaker registrada: 8.5

export class mPrenda {
  protected _codigo: string = "";
  protected _marca: string = "";
  protected _costo: number = 0;

  constructor(codigo: string, marca: string, costo: number) {
    this.codigo = codigo;
    this.marca = marca;
    this.costo = costo;
  }

  set codigo(valor: string) {
    this._codigo = valor.trim();
  }

  get codigo(): string {
    return this._codigo;
  }

  set marca(valor: string) {
    this._marca = valor.trim();
  }

  get marca(): string {
    return this._marca;
  }

  set costo(valor: number) {
    this._costo = +valor; // Coerción numérica estricta
  }

  get costo(): number {
    return this._costo;
  }

  // Margen base del 30% sobre el costo de adquisición
  precioInicial(): number {
    return this.costo * 1.30;
  }

  // Método base que puede ser sobreescrito por las subclases especializadas
  calcularPrecioFinal(): number {
    return this.precioInicial();
  }

  // Ganancia neta: precioFinal - costo
  calcularGananciaNeta(): number {
    return this.calcularPrecioFinal() - this.costo;
  }

  obtenerTipo(): string {
    return "Prenda";
  }
}

export class mSneaker extends mPrenda {

    codigo: string,
    marca: string,
    costo: number,
    esAltaDemanda: boolean,
    tallaUs: number
  ) {

    this.esAltaDemanda = esAltaDemanda;
    this.tallaUs = tallaUs;
  }

  set esAltaDemanda(valor: boolean) {
    this._esAltaDemanda = Boolean(valor);
  }

  get esAltaDemanda(): boolean {
    return this._esAltaDemanda;
  }

  set tallaUs(valor: number) {
    this._tallaUs = +valor; // Coerción numérica para talla real
  }

  get tallaUs(): number {
    return this._tallaUs;
  }

  // Sobreescritura: suma $15 de tasa de importación al precio inicial.
  // Si es talla de alta demanda, incrementa un 10% adicional sobre ese subtotal.
  override calcularPrecioFinal(): number {
    const subtotal = super.precioInicial() + 15;
    if (this.esAltaDemanda) {
      return subtotal * 1.10;
    }
    return subtotal;
  }

  override obtenerTipo(): string {
    return "Sneaker";
  }
}

import { mPrenda } from "./mPrenda.js";
import { mSneaker } from "./mSneaker.js";

export class mStore {
  // Variables de control C-A-M-P-P (Contadores, Acumuladores, Menor/Mayor)
  private acGananciaTotal: number = 0;
  private cntPrendas: number = 0;
  private mayorGanancia: number = 0;
  private codMayorGanancia: string = "";
  private minTallaSneaker: number = 0;
  private cntSneakers: number = 0;

  // Acepta mPrenda y cualquier subclase (mSneaker, mHoodie) por la regla "Es-un"
  procesarPrenda(prenda: mPrenda): void {
    const ganancia = prenda.calcularGananciaNeta();

    // 1. Acumular ganancia total y contar prendas
    this.acGananciaTotal += ganancia;
    this.cntPrendas++;

    // 2. Determinar producto con mayor ganancia neta
    if (this.cntPrendas === 1 || ganancia > this.mayorGanancia) {
      this.mayorGanancia = ganancia;
      this.codMayorGanancia = prenda.codigo;
    }

    // 3. Nuevo Requerimiento: determinar la menor talla registrada entre los sneakers vendidos
    if (prenda instanceof mSneaker) {
      this.cntSneakers++;
      if (this.cntSneakers === 1 || prenda.tallaUs < this.minTallaSneaker) {
        this.minTallaSneaker = prenda.tallaUs;
      }
    }
  }

  gananciaTotal(): number {
    return this.acGananciaTotal;
  }

  codigoMayorGanancia(): string {
    return this.codMayorGanancia;
  }

  montoMayorGanancia(): number {
    return this.mayorGanancia;
  }

  menorTallaSneaker(): number {
    return this.minTallaSneaker;
  }

  cantidadPrendas(): number {
    return this.cntPrendas;
  }

  cantidadSneakers(): number {
    return this.cntSneakers;
  }

  reiniciar(): void {
    this.acGananciaTotal = 0;
    this.cntPrendas = 0;
    this.mayorGanancia = 0;
    this.codMayorGanancia = "";
    this.minTallaSneaker = 0;
    this.cntSneakers = 0;
  }
}

import { mStore } from "../models/mStore.js";
import { mPrenda } from "../models/mPrenda.js";
import { mSneaker } from "../models/mSneaker.js";
import { mHoodie } from "../models/mHoodie.js";
import { IStoreView } from "../interfaces/IStoreView.js";

export class cStore {
  private modelo: mStore;
  private vista: IStoreView;

  // Inyección de dependencias en el constructor
  constructor(modeloIny: mStore, vistaIny: IStoreView) {
    this.modelo = modeloIny;
    this.vista = vistaIny;
  }

  iniciar(): void {
    // 1. Suscripción de eventos desde la Vista
    this.vista.onProcesar(() => this.procesarProducto());
    this.vista.onCargarCasosPrueba(() => this.cargarCasosPrueba());
    this.vista.onLimpiar(() => this.reiniciar());

    // 2. Ejecución automática de los datos de prueba oficiales al cargar
    this.cargarCasosPrueba();
  }

  private procesarProducto(): void {
    const codigo = this.vista.codigo;
    const marca = this.vista.marca;
    const costo = this.vista.costo;
    const tipo = this.vista.tipo;

    // Validaciones básicas de entrada
    if (!codigo || !marca) {
      this.vista.mostrarMensaje("Por favor, ingrese el código y la marca del producto.", "warning");
      return;
    }

    if (isNaN(costo) || costo <= 0) {
      this.vista.mostrarMensaje("El costo debe ser un valor numérico positivo mayor a 0.", "danger");
      return;
    }

    let prenda: mPrenda;

    if (tipo === "sneaker") {
      const tallaUs = this.vista.tallaUs;
      if (isNaN(tallaUs) || tallaUs <= 0) {
        this.vista.mostrarMensaje("Por favor ingrese una talla US válida (número real mayor a 0).", "danger");
        return;
      }
      prenda = new mSneaker(
        codigo,
        marca,
        costo,
        this.vista.esAltaDemanda,
        tallaUs
      );
    } else {
      prenda = new mHoodie(
        codigo,
        marca,
        costo,
        this.vista.conEstampado,
        this.vista.esColeccionPasada
      );
    }

    // 1. Procesar en el Modelo Mayor
    this.modelo.procesarPrenda(prenda);

    // 2. Reportar prenda individual en la Vista
    this.vista.reportarPrenda(
      prenda.codigo,
      prenda.marca,
      prenda.obtenerTipo(),
      prenda.calcularPrecioFinal(),
      prenda.calcularGananciaNeta()
    );

    // 3. Reporte gerencial de resumen
    this.vista.reportarResumen(
      this.modelo.gananciaTotal(),
      this.modelo.codigoMayorGanancia(),
      this.modelo.montoMayorGanancia(),
      this.modelo.menorTallaSneaker(),
      this.modelo.cantidadPrendas()
    );

    this.vista.limpiarFormulario();
    this.vista.mostrarMensaje(`Artículo ${prenda.codigo} procesado correctamente.`, "success");
  }

  private cargarCasosPrueba(): void {
    // Limpieza de estados previos
    this.modelo.reiniciar();
    this.vista.reiniciarVista();

    // 1. Instanciamos objetos individuales (estricto: cero arreglos [])
    // S01: Sneaker, Nike, Costo: 50, Alta demanda: Sí, Talla: 9.5
    const prenda1 = new mSneaker("S01", "Nike", 50, true, 9.5);

    // H01: Hoodie, Adidas, Costo: 20, Estampado: Sí, Colección pasada: No
    const prenda2 = new mHoodie("H01", "Adidas", 20, true, false);

    // S02: Sneaker, Puma, Costo: 40, Alta demanda: No, Talla: 8.5
    const prenda3 = new mSneaker("S02", "Puma", 40, false, 8.5);

    // H02: Hoodie, Vans, Costo: 30, Estampado: No, Colección pasada: Sí
    const prenda4 = new mHoodie("H02", "Vans", 30, false, true);

    // 2. Procesamos cada instancia en la Clase Mayor y reportamos en Vista
    this.modelo.procesarPrenda(prenda1);
    this.vista.reportarPrenda(
      prenda1.codigo,
      prenda1.marca,
      prenda1.obtenerTipo(),
      prenda1.calcularPrecioFinal(),
      prenda1.calcularGananciaNeta()
    );

    this.modelo.procesarPrenda(prenda2);
    this.vista.reportarPrenda(
      prenda2.codigo,
      prenda2.marca,
      prenda2.obtenerTipo(),
      prenda2.calcularPrecioFinal(),
      prenda2.calcularGananciaNeta()
    );

    this.modelo.procesarPrenda(prenda3);
    this.vista.reportarPrenda(
      prenda3.codigo,
      prenda3.marca,
      prenda3.obtenerTipo(),
      prenda3.calcularPrecioFinal(),
      prenda3.calcularGananciaNeta()
    );

    this.modelo.procesarPrenda(prenda4);
    this.vista.reportarPrenda(
      prenda4.codigo,
      prenda4.marca,
      prenda4.obtenerTipo(),
      prenda4.calcularPrecioFinal(),
      prenda4.calcularGananciaNeta()
    );

    // 3. Reporte global final desde la Clase Mayor
    this.vista.reportarResumen(
      this.modelo.gananciaTotal(),
      this.modelo.codigoMayorGanancia(),
      this.modelo.montoMayorGanancia(),
      this.modelo.menorTallaSneaker(),
      this.modelo.cantidadPrendas()
    );

    this.vista.mostrarMensaje("Casos de prueba oficiales cargados con éxito.", "info");
  }

  private reiniciar(): void {
    this.modelo.reiniciar();
    this.vista.reiniciarVista();
    this.vista.mostrarMensaje("Inventario reiniciado correctamente.", "info");
  }
}
